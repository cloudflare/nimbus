## Examples

### Basic

Three tabs with text bodies — triggers are synthesised from each `label` prop.

<Example code={`<Tabs>
  <TabItem label="One">First panel content.</TabItem>
  <TabItem label="Two">Second panel content.</TabItem>
  <TabItem label="Three">Third panel content.</TabItem>
</Tabs>`} contentClass="w-full">
  <div class="w-full">
    <Tabs>
      <TabItem label="One">First panel content.</TabItem>
      <TabItem label="Two">Second panel content.</TabItem>
      <TabItem label="Three">Third panel content.</TabItem>
    </Tabs>
  </div>
</Example>

### Synced

Two Tabs instances with the same `syncKey` mirror each other — pick a tab once, every matching instance follows.

<Example code={`<Tabs syncKey="lang">
  <TabItem label="TypeScript">type User = { id: string };</TabItem>
  <TabItem label="JavaScript">// inferred at runtime</TabItem>
</Tabs>

<Tabs syncKey="lang">
  <TabItem label="TypeScript">function load(id: string) { ... }</TabItem>
  <TabItem label="JavaScript">function load(id) { ... }</TabItem>
</Tabs>`} contentClass="w-full">
  <div class="w-full flex flex-col gap-4">
    <Tabs syncKey="lang-demo">
      <TabItem label="TypeScript"><code>type User = {`{ id: string }`};</code></TabItem>
      <TabItem label="JavaScript"><code>// inferred at runtime</code></TabItem>
    </Tabs>
    <Tabs syncKey="lang-demo">
      <TabItem label="TypeScript"><code>function load(id: string) {`{ ... }`}</code></TabItem>
      <TabItem label="JavaScript"><code>function load(id) {`{ ... }`}</code></TabItem>
    </Tabs>
  </div>
</Example>

### Manual

For triggers you write yourself, use `TabsList`, `TabsTrigger` and `TabsContent`. Each trigger opens the panel with the same `value`, whatever order the panels are in. Text before `TabsList` stays above the tabs.

<Example code={`<Tabs>
  <TabsList>
    <TabsTrigger value="npm">npm</TabsTrigger>
    <TabsTrigger value="pnpm">pnpm</TabsTrigger>
  </TabsList>
  <TabsContent value="npm">npm install @cloudflare/nimbus-docs</TabsContent>
  <TabsContent value="pnpm">pnpm add @cloudflare/nimbus-docs</TabsContent>
</Tabs>`} contentClass="w-full">
  <div class="w-full">
    <Tabs>
      <TabsList>
        <TabsTrigger value="npm">npm</TabsTrigger>
        <TabsTrigger value="pnpm">pnpm</TabsTrigger>
      </TabsList>
      <TabsContent value="npm"><code>npm install @cloudflare/nimbus-docs</code></TabsContent>
      <TabsContent value="pnpm"><code>pnpm add @cloudflare/nimbus-docs</code></TabsContent>
    </Tabs>
  </div>
</Example>
